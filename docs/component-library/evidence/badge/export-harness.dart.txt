ThemeData neutral(bool dark) {
  final base = dark ? AppTheme.dark : AppTheme.light;
  final bg = Color(dark ? 0xff0a0a0a : 0xffffffff);
  final fg = Color(dark ? 0xfffafafa : 0xff000000);
  final muted = Color(dark ? 0xff262626 : 0xfff5f5f5);
  final colors = base.colorScheme.copyWith(
    surface: bg,
    onSurface: fg,
    primary: Color(dark ? 0xffe5e5e5 : 0xff000000),
    onPrimary: Color(dark ? 0xff171717 : 0xfffafafa),
    outlineVariant: dark ? const Color(0x1affffff) : const Color(0xffe5e5e5),
    error: Color(dark ? 0xffff6467 : 0xffe7000b),
  );
  return base.copyWith(
    colorScheme: colors,
    scaffoldBackgroundColor: bg,
    extensions: [
      ...base.extensions.values.where((x) => x is! DTokens),
      ReferenceTokens(
        colors: colors,
        background: bg,
        surface: bg,
        muted: muted,
        border: dark ? const Color(0x1affffff) : const Color(0xffe5e5e5),
        hover: muted,
        selected: muted,
        selectedForeground: fg,
        radius: 10,
      ),
    ],
  );
}

class ReferenceTokens extends DTokens {
  const ReferenceTokens({
    required super.colors,
    required super.background,
    required super.surface,
    required super.muted,
    required super.border,
    required super.hover,
    required super.selected,
    required super.selectedForeground,
    super.radius,
  });
  @override
  Color get focusRing => colors.brightness == Brightness.dark
      ? const Color(0xff737373)
      : const Color(0xffa1a1a1);
  @override
  Color get mutedForeground => colors.brightness == Brightness.dark
      ? const Color(0xffa1a1a1)
      : const Color(0xff737373);
}
